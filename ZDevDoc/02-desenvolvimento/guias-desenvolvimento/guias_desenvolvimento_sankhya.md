```sql
-- Trigger para validação de registros
CREATE OR REPLACE TRIGGER TRG_VALIDAR_REGISTRO_PONTO
    BEFORE INSERT OR UPDATE ON AD_REGISTRO_PONTO
    FOR EACH ROW
DECLARE
    P_JORNADA AD_JORNADA_TRABALHO%ROWTYPE;
    P_ULTIMO_REGISTRO AD_REGISTRO_PONTO%ROWTYPE;
    P_HORARIO_VALIDO BOOLEAN := TRUE;
BEGIN
    -- Obter dados da jornada
    SELECT * INTO P_JORNADA
    FROM AD_JORNADA_TRABALHO
    WHERE ID_JORNADA = :NEW.ID_JORNADA
    AND ATIVO = 'S';
    
    -- Validar horário de entrada
    IF :NEW.TIPO_REGISTRO = 'ENTRADA' THEN
        IF :NEW.HORARIO_REGISTRO < (P_JORNADA.HORARIO_ENTRADA - P_JORNADA.TOLERANCIA_MINUTOS/1440) THEN
            P_HORARIO_VALIDO := FALSE;
        END IF;
    END IF;
    
    -- Validar horário de saída
    IF :NEW.TIPO_REGISTRO = 'SAIDA' THEN
        IF :NEW.HORARIO_REGISTRO > (P_JORNADA.HORARIO_SAIDA + P_JORNADA.TOLERANCIA_MINUTOS/1440) THEN
            P_HORARIO_VALIDO := FALSE;
        END IF;
    END IF;
    
    -- Aplicar validação
    IF NOT P_HORARIO_VALIDO THEN
        :NEW.STATUS := 'PENDENTE_APROVACAO';
        :NEW.OBSERVACOES := 'Horário fora da tolerância permitida';
    END IF;
    
EXCEPTION
    WHEN OTHERS THEN
        RAISE_APPLICATION_ERROR(-20001, 'Erro na validação: ' || SQLERRM);
END;
/
```

```sql
-- Procedure para verificar vencimentos
CREATE OR REPLACE PROCEDURE SP_VERIFICAR_VENCIMENTOS_CONTRATOS AS
    P_DIAS_ANTECEDENCIA NUMBER := 30;
    P_COUNT_ALERTAS NUMBER := 0;
    
    CURSOR C_CONTRATOS_VENCENDO IS
        SELECT 
            ID_CONTRATO,
            NUMERO,
            PARTE_CONTRATANTE,
            PARTE_CONTRATADA,
            DATA_FIM,
            (DATA_FIM - SYSDATE) AS DIAS_PARA_VENCIMENTO
        FROM AD_CONTRATOS
        WHERE STATUS = 'ATIVO'
        AND DATA_FIM BETWEEN SYSDATE AND (SYSDATE + P_DIAS_ANTECEDENCIA)
        AND ID_CONTRATO NOT IN (
            SELECT ID_CONTRATO 
            FROM AD_ALERTAS_CONTRATO 
            WHERE TIPO_ALERTA = 'VENCIMENTO'
            AND TRUNC(DATA_ALERTA) = TRUNC(SYSDATE)
        );
        
BEGIN
    FOR C IN C_CONTRATOS_VENCENDO LOOP
        -- Inserir alerta de vencimento
        INSERT INTO AD_ALERTAS_CONTRATO (
            ID_CONTRATO,
            TIPO_ALERTA,
            MENSAGEM,
            STATUS
        ) VALUES (
            C.ID_CONTRATO,
            'VENCIMENTO',
            'Contrato ' || C.NUMERO || ' vence em ' || 
            ROUND(C.DIAS_PARA_VENCIMENTO) || ' dias. ' ||
            'Contratante: ' || C.PARTE_CONTRATANTE || 
            ' | Contratada: ' || C.PARTE_CONTRATADA,
            'PENDENTE'
        );
        
        P_COUNT_ALERTAS := P_COUNT_ALERTAS + 1;
    END LOOP;
    
    -- Log da execução
    INSERT INTO AD_LOG_EXECUCAO (
        PROCEDURE,
        STATUS,
        OBSERVACAO,
        DT_EXECUCAO
    ) VALUES (
        'SP_VERIFICAR_VENCIMENTOS_CONTRATOS',
        'CONCLUIDO',
        'Alertas gerados: ' || P_COUNT_ALERTAS,
        SYSDATE
    );
    
    COMMIT;
    
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        INSERT INTO AD_LOG_ERRO (
            PROCEDURE,
            ERRO,
            DT_ERRO
        ) VALUES (
            'SP_VERIFICAR_VENCIMENTOS_CONTRATOS',
            SQLERRM,
            SYSDATE
        );
        COMMIT;
END;
/
```

```sql
    FROM AD_OPERACAO_TRIANGULAR
```

```java
// Serviço de integração
@Service
public class IntegracaoService {
    
    private Map<String, IntegradorApp> integradores;
    
    @PostConstruct
    public void inicializar() {
        integradores = new HashMap<>();
        
        // Registrar integradores
        integradores.put("CRM", new IntegradorCRM());
        integradores.put("ESTOQUE", new IntegradorEstoque());
        integradores.put("FINANCEIRO", new IntegradorFinanceiro());
    }
    
    public Map<String, Object> sincronizarDados(String tipoDados, Map<String, Object> dados) {
        IntegradorApp integrador = integradores.get(tipoDados);
        
        if (integrador == null) {
            throw new IllegalArgumentException("Tipo de dados não suportado: " + tipoDados);
        }
        
        return integrador.sincronizar(dados);
    }
    
    public Map<String, Object> obterStatusIntegracao(String aplicativo) {
        IntegradorApp integrador = integradores.get(aplicativo);
        
        if (integrador == null) {
            throw new IllegalArgumentException("Aplicativo não encontrado: " + aplicativo);
        }
        
        return integrador.obterStatus();
    }
}
```
